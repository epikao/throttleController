/*****************************************************/
void Write_REG_Index(unsigned char reg_index)
{
	unsigned char i;

	pLCD_CS =0;
	pLCD_RD=1;
	pLCD_RS=0;

	for(i=0; i<8; i++)
	{
	 	if(reg_index&0x80)
			pLCD_SDA=1;
		else
			pLCD_SDA=0;

		_nop_();
		pLCD_SCL=0;
		_nop_();
		_nop_();
		_nop_();
		_nop_();     
		pLCD_SCL=1;
		_nop_();
		_nop_();

		reg_index <<=1;
	}

	pLCD_CS=1;   
}
/*****************************************************/
void Write_REG_Data(unsigned char reg_data)
{
	unsigned char i;

	pLCD_CS=0;
	pLCD_RD=1;
	pLCD_RS=1;

	for(i=0; i<8; i++)
	{
	 	if(reg_data&0x80)
			pLCD_SDA=1;
		else
			pLCD_SDA=0;

		_nop_();
		pLCD_SCL=0;
		_nop_();
		_nop_();
		_nop_();
		_nop_();     
		pLCD_SCL=1;
		_nop_();
		_nop_();

		reg_data <<=1;
	}

	pLCD_CS=1;  

}

/********************************************/
void Driver_Initial(void)
{

	pLCD_RST = 0;
	Delay(2000);
	
	pLCD_RST = 1;
	Delay(2000);


	
	Write_REG_Index(0x11);	// Exit Sleep 
	Delay(600); 

	Write_REG_Index(0x36);	// Memory Access Control 
	Write_REG_Data(0x00);	

	Write_REG_Index(0x3a);	
	Write_REG_Data(0x66);	    
	
	Write_REG_Index(0xb2);  
	Write_REG_Data(0x0c); 
	Write_REG_Data(0x0c); 
	Write_REG_Data(0X00); 
	Write_REG_Data(0X33); 
	Write_REG_Data(0X33);

	
	Write_REG_Index(0xb7);  //VGH VGL
	Write_REG_Data(0X35);    //
	
	Write_REG_Index(0xbb); //VCOM
	Write_REG_Data(0X20);	//28

	Write_REG_Index(0xc0); 
	Write_REG_Data(0X2c);	
		
	Write_REG_Index(0xc2); //
	Write_REG_Data(0X01);
	Write_REG_Data(0Xff);	
	
	Write_REG_Index(0xc3); //VRH  
	Write_REG_Data(0X0b);
	
	Write_REG_Index(0xc4); //vdv
	Write_REG_Data(0X20);
	
   	Write_REG_Index(0xc6); //frame rate
   	Write_REG_Data(0X0f);

	Write_REG_Index(0xd0); 
	Write_REG_Data(0Xa4);
	Write_REG_Data(0X51);


	Write_REG_Index(0x2A); //Column address
	Write_REG_Data(0X00);      
	Write_REG_Data(0X00);   
	Write_REG_Data(0X00);   //0x00ef=239
	Write_REG_Data(0Xef);   
	Write_REG_Index(0x2b); //row address 
	Write_REG_Data(0X00);      
	Write_REG_Data(0X00);   
	Write_REG_Data(0X01);   //0x013f=319
	Write_REG_Data(0X3f);   
	
	Write_REG_Index(0xE0);	// Set Gamma 
	Write_REG_Data(0xd0); 
	Write_REG_Data(0x00);
	Write_REG_Data(0x02);
	Write_REG_Data(0x07); 
	Write_REG_Data(0x0a); 
	Write_REG_Data(0x28);
	Write_REG_Data(0x32);
	Write_REG_Data(0x44); 
	Write_REG_Data(0x42); 
	Write_REG_Data(0x06); 
	Write_REG_Data(0x0e); 
	Write_REG_Data(0x12); 
	Write_REG_Data(0x14);
	Write_REG_Data(0x17);
	 
	Write_REG_Index(0XE1);	// Set Gamma 
	Write_REG_Data(0xd0); 
	Write_REG_Data(0x00);
	Write_REG_Data(0x02);
	Write_REG_Data(0x07); 
	Write_REG_Data(0x0a); 
	Write_REG_Data(0x28); 
	Write_REG_Data(0x31); 
	Write_REG_Data(0x54); 
	Write_REG_Data(0x47); 
	Write_REG_Data(0x0e); 
	Write_REG_Data(0x1c); 
	Write_REG_Data(0x17); 
	Write_REG_Data(0x1b); 
	Write_REG_Data(0x1e); 
	
	Write_REG_Index(0x20);	

	Delay(1000); 
	Write_REG_Index(0x29);	// Display on 
	Delay(1000); 
}
